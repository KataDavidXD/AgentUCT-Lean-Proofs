% Shared presentation for standalone supplementary and mathematical-proof readers.
% The canonical mathematical and experimental content remains in its source files.
\newcommand{\AgentUCTReading}{}
\usepackage[a4paper,margin=22mm]{geometry}
\usepackage[T1]{fontenc}
\usepackage{libertine}
\usepackage{inconsolata}
\usepackage{amsmath,amsthm}
\usepackage[libertine]{newtxmath}
\input{paper/preamble}
\usepackage{xcolor}
\usepackage{fancyhdr}
\usepackage{xurl}
\usepackage{hyperref}
\usepackage{etoolbox}
\usepackage{placeins}
\hypersetup{colorlinks=true,linkcolor=blue!45!black,urlcolor=blue!45!black,
  citecolor=blue!45!black,pdfauthor={}}
\setlength{\headheight}{14pt}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small AgentUCT}
\fancyhead[R]{\small\ReaderShortTitle}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.3pt}
\renewcommand{\footrulewidth}{0pt}
\setlength{\parindent}{1em}
\setlength{\parskip}{2pt}
\raggedbottom
\setlength{\emergencystretch}{2em}
\renewcommand{\topfraction}{0.95}
\renewcommand{\bottomfraction}{0.95}
\renewcommand{\textfraction}{0.05}
\renewcommand{\floatpagefraction}{0.85}
\setcounter{topnumber}{5}
\setcounter{bottomnumber}{5}
\setcounter{totalnumber}{8}
\setcounter{equation}{0}
\setcounter{table}{0}
\setcounter{figure}{0}
\makeatletter
% Keep the title compact and float-only pages aligned at the top.
\renewcommand{\maketitle}{%
  \begin{center}\vspace*{3mm}{\Large\@title\par}\vspace{3mm}\end{center}}
\setlength{\@fptop}{0pt}
\setlength{\@fpsep}{12pt}
\setlength{\@fpbot}{0pt plus 1fil}
% The journal layout uses p-only tables. In a one-column reader allow them
% beside their discussion, instead of forcing a nearly empty float page.
\let\ReaderOriginalTable\table
\let\ReaderOriginalEndTable\endtable
\renewenvironment{table}[1][htbp]{\ReaderOriginalTable[htbp]}{\ReaderOriginalEndTable}
\renewenvironment{table*}[1][htbp]{\ReaderOriginalTable[htbp]}{\ReaderOriginalEndTable}
\AtBeginEnvironment{table}{\renewcommand{\arraystretch}{1.1}}
\AtBeginEnvironment{table*}{\renewcommand{\arraystretch}{1.1}}
\pretocmd{\section}{\FloatBarrier}{}{}
\AtBeginDocument{%
\let\ReaderOriginalRef\ref
\renewcommand{\ref}[1]{%
  \ifcsname r@#1\endcsname
    \ReaderOriginalRef{#1}%
  \else
    \ifcsname r@reader-external:#1\endcsname
      \ReaderOriginalRef*{reader-external:#1}%
    \else
      \ReaderOriginalRef{#1}%
    \fi
  \fi}}
\makeatother
